## React Hook

```typescript
import { useState, useEffect } from "react";

export function useSubscription(subscriptionId: string) {
  const [subscription, setSubscription] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    async function fetchSubscription() {
      try {
        const response = await fetch(
          `https://api.stellar-subscriptions.io/subscriptions/${subscriptionId}`
        );
        const data = await response.json();
        setSubscription(data.subscription);
      } catch (err) {
        setError(err);
      } finally {
        setLoading(false);
      }
    }
    
    fetchSubscription();
  }, [subscriptionId]);
  
  return { subscription, loading, error };
}
```

  private sdk: StellarSubscriptions;
  private merchantKeypair: Keypair;
  private intervalMs: number;
  
  constructor(sdk: StellarSubscriptions, merchantKeypair: Keypair, intervalMs = 600000) {
    this.sdk = sdk;
    this.merchantKeypair = merchantKeypair;
    this.intervalMs = intervalMs;
  }
  
  async start() {
    console.log("Starting charging worker...");
    while (true) {
      await this.runChargeCycle();
      await new Promise(resolve => setTimeout(resolve, this.intervalMs));
    }
  }

    try {
      const chargeable = await this.getChargeableSubscriptions();
      console.log(`Found ${chargeable.length} subscriptions to charge`);

## Best Practices

<CalloutBox type="success">
**Security:**
- Never expose private keys in frontend code
- Use environment variables for secrets
- Validate all inputs before sending transactions
- Implement rate limiting on API calls
</CalloutBox>

<CalloutBox type="info">
**Performance:**
- Cache API responses when appropriate
- Batch multiple subscription queries
- Use webhooks or event streams for real-time updates
- Implement exponential backoff for retries
</CalloutBox>
